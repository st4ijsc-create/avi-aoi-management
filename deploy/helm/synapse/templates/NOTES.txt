2. Liveness / readiness endpoint (real): GET /health on port {{ .Values.service.port }}
{{- if .Values.ingress.enabled }}
   Reachable via Ingress:
   {{- range .Values.ingress.hosts }}
     http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}/health
   {{- end }}
{{- else if eq .Values.service.type "NodePort" }}
     export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[?(@.type=="InternalIP")].address}')
     export NODE_PORT=$(kubectl -n {{ .Release.Namespace }} get svc {{ include "synapse.fullname" . }} -o jsonpath='{.spec.ports[0].nodePort}')
     curl http://$NODE_IP:$NODE_PORT/health
{{- else }}
   Port-forward and curl:
     kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "synapse.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}
